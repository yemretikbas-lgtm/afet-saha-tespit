async function normalizeImage(file){

 // Saha arşivi + Excel raporu için dengeli sıkıştırma: uzun kenar en çok 1600 px,
 // hedef yaklaşık 420 KB/foto. Böylece 50-100 rapor fotoğrafında Excel gereksiz büyümez.
 const max=1600,scale=Math.min(1,max/Math.max(iw,ih)),w=Math.max(1,Math.round(iw*scale)),h=Math.max(1,Math.round(ih*scale));
 const c=document.createElement('canvas');c.width=w;c.height=h;const ctx=c.getContext('2d',{alpha:false});ctx.drawImage(img,0,0,w,h);img.close?.();
 const target=420*1024;let q=.78,blob=await canvasJpegBlob(c,q);
 while(blob.size>target && q>.54){q=Math.max(.54,q-.06);blob=await canvasJpegBlob(c,q);}
 return {id:uid(),blob,width:w,height:h,selected:false,capturedAt:new Date().toISOString(),coordX:valueNum(els.x),coordY:valueNum(els.y),gpsAccuracy:valueNum(els.gps.dataset.accuracy)};
}

function blobToDataUrl(blob){return new Promise((res,rej)=>{const fr=new FileReader();fr.onload=()=>res(fr.result);fr.onerror=()=>rej(fr.error);fr.readAsDataURL(blob);});}
async function serializeRecords(rows){
 const serial=[];
 for(const r of rows){
   const rr={...r,photos:[]};
   for(const p of (r.photos||[])){
     const data=await blobToDataUrl(p.blob);
     rr.photos.push({...p,blob:undefined,data});
   }
   serial.push(rr);
 }
 return serial;
}
